import { Aside, Steps, Tabs, TabItem } from '@astrojs/starlight/components';

This example builds a durable workflow that takes a US ZIP code, resolves it to a city via the [Zippopotamus API](https://www.zippopotam.us/), then fetches breweries in that city from the [Open Brewery DB](https://www.openbrewerydb.org/). All steps are persisted — if the process crashes mid-flight, completed API calls are not retried.

**Features demonstrated:** `#[zart_durable]`, `#[zart_step]`, sequential steps, structured output.

## Step definitions

Each step is defined as a standalone async function. Zart persists each step's result before moving on.

<Tabs syncKey="sdk">
<TabItem label="Rust">

```rust
use zart::error::StepError;
use zart::prelude::*;
use zart::zart_step;

#[zart_step("lookup-zip", retry = "exponential(3, 1s)")]
async fn lookup_zip(
    client: &reqwest::Client,
    zip_code: &str,
) -> Result<(String, String), StepError> {
    println!("[lookup-zip] Attempt {}", zart::context().current_attempt + 1);

    let resp = client
        .get(format!("https://api.zippopotam.us/us/{zip_code}"))
        .send().await
        .map_err(|e| StepError::Failed {
            step: "lookup-zip".into(), reason: e.to_string(),
        })?;

    let zip_resp: ZipResponse = resp.json().await
        .map_err(|e| StepError::Failed {
            step: "lookup-zip".into(), reason: format!("parse error: {e}"),
        })?;

    let place = zip_resp.places.first().ok_or_else(|| StepError::Failed {
        step: "lookup-zip".into(),
        reason: format!("no place found for ZIP {zip_code}"),
    })?;

    Ok((place.place_name.clone(), place.state.clone()))
}

#[zart_step("find-breweries", retry = "exponential(3, 1s)")]
async fn find_breweries(
    client: &reqwest::Client,
    city: &str,
) -> Result<Vec<BreweryRaw>, StepError> {
    println!("[find-breweries] Attempt {}", zart::context().current_attempt + 1);

    let resp = client
        .get("https://api.openbrewerydb.org/v1/breweries")
        .query(&[("by_city", city)])
        .send().await
        .map_err(|e| StepError::Failed {
            step: "find-breweries".into(), reason: e.to_string(),
        })?;

    resp.json().await.map_err(|e| StepError::Failed {
        step: "find-breweries".into(),
        reason: format!("parse error: {e}"),
    })
}

#[zart_step("transform-results")]
async fn transform_results(
    raw: &[BreweryRaw],
    city: &str,
    state: &str,
) -> Result<Vec<BreweryInfo>, StepError> {
    Ok(raw.iter().map(|b| BreweryInfo {
        name: b.name.clone(),
        brewery_type: b.brewery_type.clone().unwrap_or_else(|| "unknown".into()),
        city: b.city.clone().unwrap_or_else(|| city.to_string()),
        state: b.state.clone().unwrap_or_else(|| state.to_string()),
    }).collect())
}
```

```python
import httpx
import zart

@zart.step("lookup-zip", retry=zart.Retry.exponential(3, "1s"))
async def lookup_zip(zip_code: str) -> tuple[str, str]:
    print(f"[lookup-zip] Attempt {zart.context().current_attempt + 1}")
    async with httpx.AsyncClient() as client:
        resp = await client.get(f"https://api.zippopotam.us/us/{zip_code}")
        resp.raise_for_status()
        places = resp.json().get("places", [])
        if not places:
            raise ValueError(f"no place found for ZIP {zip_code}")
        return places[0]["place name"], places[0]["state"]

@zart.step("find-breweries", retry=zart.Retry.exponential(3, "1s"))
async def find_breweries(city: str) -> list[dict]:
    print(f"[find-breweries] Attempt {zart.context().current_attempt + 1}")
    async with httpx.AsyncClient() as client:
        resp = await client.get(
            "https://api.openbrewerydb.org/v1/breweries",
            params={"by_city": city},
        )
        resp.raise_for_status()
        return resp.json()

@zart.step("transform-results")
async def transform_results(
    raw: list[dict],
    city: str,
    state: str,
) -> list[BreweryInfo]:
    return [
        BreweryInfo(
            name=b["name"],
            brewery_type=b.get("brewery_type") or "unknown",
            city=b.get("city") or city,
            state=b.get("state") or state,
        )
        for b in raw
    ]
```

```ts
import { Effect } from "effect";
import { Zart } from "@zart/effect";

const lookupZip = Zart.step(
  "lookup-zip",
  (zipCode: string) =>
    Effect.gen(function* () {
      console.log(`[lookup-zip] Resolving ZIP ${zipCode}`);
      const data = yield* Effect.tryPromise(async () => {
        const resp = await fetch(`https://api.zippopotam.us/us/${zipCode}`);
        if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
        return (await resp.json()) as { places: Array<{ "place name": string; state: string }> };
      });
      const place = data.places[0];
      if (!place) return yield* Effect.fail(new Error(`no place found for ZIP ${zipCode}`));
      return { city: place["place name"], state: place.state };
    }),
  { retry: Zart.Retry.exponential(3, "1 seconds") },
);

const findBreweries = Zart.step(
  "find-breweries",
  (city: string) =>
    Effect.gen(function* () {
      console.log(`[find-breweries] Querying city ${city}`);
      return yield* Effect.tryPromise(async () => {
        const url = new URL("https://api.openbrewerydb.org/v1/breweries");
        url.searchParams.set("by_city", city);
        const resp = await fetch(url);
        if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
        return (await resp.json()) as Array<{
          name: string;
          brewery_type?: string;
          city?: string;
          state?: string;
        }>;
      });
    }),
  { retry: Zart.Retry.exponential(3, "1 seconds") },
);

const transformResults = Zart.step(
  "transform-results",
  (
    raw: Array<{ name: string; brewery_type?: string; city?: string; state?: string }>,
    city: string,
    state: string,
  ) =>
    Effect.succeed(
      raw.map((b) => ({
        name: b.name,
        breweryType: b.brewery_type ?? "unknown",
        city: b.city ?? city,
        state: b.state ?? state,
      })),
    ),
);
```

## The workflow

The durable handler composes steps sequentially. Each step call is persisted before moving on.

<Tabs syncKey="sdk">
<TabItem label="Rust">

```rust
#[zart_durable("brewery-finder", timeout = "5m")]
async fn brewery_finder(data: FinderInput) -> Result<FinderOutput, TaskError> {
    let client = reqwest::Client::new();

    // Step 1: resolve ZIP → city/state, retry up to 3× with exponential backoff
    let (city, state) = lookup_zip(&client, &data.zip_code).await?;

    // Step 2: find breweries in that city, retry on transient failures
    let raw_breweries = find_breweries(&client, &city).await?;

    // Step 3: transform raw API data into structured output (no retries needed)
    let breweries = transform_results(&raw_breweries, &city, &state).await?;

    Ok(FinderOutput {
        zip_code: data.zip_code,
        city,
        state,
        breweries,
        found_at: Utc::now().to_rfc3339(),
    })
}
```

```python
    return FinderOutput(
        zip_code=data.zip_code,
        city=city,
        state=state,
        breweries=breweries,
        found_at=datetime.now(UTC).isoformat(),
    )
```

```ts
        foundAt: foundAt.toISOString(),
```

<Aside type="tip">
Step results are stored in PostgreSQL after each successful step. Kill the process after step 1 completes and restart — the workflow picks up at step 2 without calling the ZIP API again.
</Aside>

## Key concepts

**`#[zart_step]` macro** — turns an async function into a step struct that implements `ZartStep`. The step name and retry config are specified as attributes. Each parameter becomes a field on the generated struct.

**`retry` attribute** — `"exponential(3, 1s)"` generates a `RetryConfig::exponential(3, Duration::from_secs(1))`. Use `"fixed(n, dur)"` for constant-delay retries. Omit the attribute for steps that don't need retry.

**Direct `.await`** — steps produced by the macro implement `IntoFuture`, so calling `step_fn(args).await?` runs the step through the framework. The step name and retry policy are applied automatically.

**Sequential data flow** — each step's return value is available to the next step as a normal Rust variable. There is no special API for passing data between steps.
